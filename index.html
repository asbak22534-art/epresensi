Kode Backend GAS Teroptimasi (syncAndUpgradeSheets — Aman Tanpa Hapus Data)

                            

                                Salin kode Code.gs di bawah ini ke editor Extensions → Apps Script pada Google Sheet Anda. Jalankan fungsi syncAndUpgradeSheets() untuk menambahkan kolom atau baris konfigurasi yang belum ada secara otomatis tanpa menghapus data siswa, kegiatan, maupun riwayat absensi lama Anda.
                            

                        

                        
                    


                    

                        

                            1. Buat / Update Kolom Sheet
                            Buka Extensions → Apps Script. Tempel kode di bawah, pilih fungsi syncAndUpgradeSheets lalu klik Run (aman untuk data lama).
                        

                        

                            2. Deploy / Manage Deployment
                            Klik Deploy → Manage deployments → Edit (Pensil) → New version → Deploy agar URL Web App tidak berubah.
                        

                        

                            3. Klik "Update Kolom Sheet"
                            Anda juga dapat menekan tombol Update Kolom Sheet di atas kapan saja untuk menyinkronkan kolom baru dari aplikasi.

/**

function setupAntrianKelompokSheet(ss) {
  let qSheet = ss.getSheetByName(SHEET_NAMES.ANTRIAN);
  if (!qSheet) {

  }
  qSheet.setTabColor('#8b5cf6');
  qSheet.setFrozenRows(3);

  const groups = [
    { name: 'Kelompok 1', startCol: 1, headerBg: '#4f46e5', subBg: '#eef2ff' },
    { name: 'Kelompok 2', startCol: 6, headerBg: '#059669', subBg: '#ecfdf5' },
    { name: 'Kelompok 3', startCol: 11, headerBg: '#d97706', subBg: '#fffbeb' },
    { name: 'Kelompok 4', startCol: 16, headerBg: '#e11d48', subBg: '#fff1f2' }
  ];

  qSheet.setRowHeight(1, 38);
  qSheet.setRowHeight(2, 26);
  qSheet.setRowHeight(3, 30);

  groups.forEach(g => {
    const c = g.startCol;
    // Banner Nama Kelompok
    const titleRange = qSheet.getRange(1, c, 1, 4);
    titleRange.merge()
      .setValue('📋 ANTRIAN ' + g.name.toUpperCase())
      .setBackground(g.headerBg)
      .setFontColor('#ffffff')
      .setFontWeight('bold')
      .setFontSize(11)
      .setHorizontalAlignment('center')
      .setVerticalAlignment('middle');

    // Baris Total Hadir Kelompok
    qSheet.getRange(2, c, 1, 2).merge()
      .setValue('Total Hadir ' + g.name + ':')
      .setBackground(g.subBg)
      .setFontWeight('bold')
      .setFontSize(9)
      .setHorizontalAlignment('right');
    qSheet.getRange(2, c + 2, 1, 2).merge()
      .setFormula('=COUNTIF(Log_Kehadiran!$L$2:$L, "' + g.name + '") & " Peserta"')
      .setBackground(g.subBg)
      .setFontWeight('bold')
      .setFontSize(10)
      .setHorizontalAlignment('center');

    // Header Kolom Tabel Antrian
    qSheet.getRange(3, c, 1, 4)
      .setValues([['No Antrian', 'Nama Siswa', 'Waktu Hadir', 'NISN']])
      .setBackground('#1e293b')
      .setFontColor('#ffffff')
      .setFontWeight('bold')
      .setFontSize(9)
      .setHorizontalAlignment('center')
      .setVerticalAlignment('middle');

    // Formula Live QUERY Urutan Antrian Berdasarkan Waktu Kehadiran
    const formula = '=IFERROR(QUERY(Log_Kehadiran!$A$2:$P, "SELECT M, G, C, F WHERE L = \'' + g.name + '\' ORDER BY M ASC", 0), {"-", "Belum ada peserta hadir", "-", "-"})';
    qSheet.getRange(4, c).setFormula(formula);

    // Atur Lebar Kolom & Pemisah
    qSheet.setColumnWidth(c, 85);
    qSheet.setColumnWidth(c + 1, 195);
    qSheet.setColumnWidth(c + 2, 100);
    qSheet.setColumnWidth(c + 3, 115);
    qSheet.setColumnWidth(c + 4, 24);
  });
}

/**
 * 2. HANDLER HTTP GET & POST (REST API UNTUK FRONTEND)
 */
function doGet(e) {
  const action = (e && e.parameter && e.parameter.action) || 'ping';
  if (action === 'initData') {
    return jsonResponse(getAllInitialData());
  }
  if (action === 'scanAttendance' && e.parameter.payload) {
    const payload = JSON.parse(e.parameter.payload);
    return jsonResponse(handleScanAttendance(payload));
  }
  return jsonResponse({ status: 'ok', message: 'E-Presensi QR GAS Backend Active' });
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;

    if (action === 'upgradeSheets') {
      return jsonResponse(syncAndUpgradeSheets());
    }
    if (action === 'scanAttendance') {
      return jsonResponse(handleScanAttendance(payload));
    }
    if (action === 'addSiswa') {
      return jsonResponse(addSiswaRow(payload.siswa));
    }
    if (action === 'addKegiatan') {
      return jsonResponse(addKegiatanRow(payload.kegiatan));
    }
    if (action === 'updateKegiatan') {
      return jsonResponse(updateKegiatanRow(payload.kegiatan));
    }
    if (action === 'setActiveKegiatan') {
      return jsonResponse(setActiveKegiatanRow(payload.idKegiatan));
    }
    if (action === 'saveConfig') {
      return jsonResponse(saveConfigSheet(payload.config));
    }
    return jsonResponse({ status: 'error', message: 'Action tidak dikenali' });
  } catch (err) {
    return jsonResponse({ status: 'error', message: err.toString() });
  }
}

/**
 * 3. PENCATATAN PRESENSI & PENOMORAN ANTRIAN 2 DIGIT OTOMATIS PER KELOMPOK
 */
function handleScanAttendance(payload) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let logSheet = ss.getSheetByName(SHEET_NAMES.LOG);
    if (!logSheet) {
      initializeSheets();
      logSheet = ss.getSheetByName(SHEET_NAMES.LOG);
    }

    const kelompokName = String(payload.kelompok || 'Kelompok 1').trim();
    const idKegiatan = String(payload.idKegiatan || 'KEG-001').trim();

    // Hitung urutan nomor antrian pada Kegiatan & Kelompok yang sama
    let queueCounter = 1;
    const lastRow = logSheet.getLastRow();
    if (lastRow > 1) {
      const existingLogs = logSheet.getRange(2, 4, lastRow - 1, 9).getDisplayValues();
      for (let i = 0; i < existingLogs.length; i++) {
        const rowKegId = String(existingLogs[i][0]).trim();
        const rowKelompok = String(existingLogs[i][8]).trim();
        if (rowKegId === idKegiatan && rowKelompok.toLowerCase() === kelompokName.toLowerCase()) {
          queueCounter++;
        }
      }
    }
    const noAntrianFormatted = String(queueCounter).padStart(2, '0');

    const now = new Date();
    const waktuWIB = Utilities.formatDate(now, 'Asia/Jakarta', 'HH:mm') + ' WIB';
    const idLog = 'LOG-' + now.getTime();

    let statusNotif = 'Terkirim via Antrean';
    if (payload.sendWa && payload.waToken && payload.noWa) {
      statusNotif = sendWhatsAppGateway(payload.waProvider, payload.waToken, payload.noWa, payload.waMessage);
    }
    if (payload.sendTg && payload.tgToken && payload.tgChatId) {
      const tgStatus = sendTelegramBotGateway(payload.tgToken, payload.tgChatId, payload.tgMessage || payload.waMessage, payload.noWa, payload.namaSiswa);
      statusNotif = statusNotif === 'Terkirim via Antrean' ? tgStatus : (statusNotif + ' | ' + tgStatus);
    }

    logSheet.appendRow([
      idLog,
      now.toISOString(),
      payload.waktu || waktuWIB,
      idKegiatan,
      payload.namaKegiatan,
      "'" + payload.nisn,
      payload.namaSiswa,
      payload.panggilan || '',
      payload.jk || 'Laki-laki',
      payload.kelas,
      payload.statusPeserta || 'CALON SISWA',
      kelompokName,
      "'" + noAntrianFormatted,
      payload.statusKehadiran,
      payload.noWa,
      statusNotif
    ]);

    return {
      status: 'success',
      idLog: idLog,
      waktu: payload.waktu || waktuWIB,
      kelompok: kelompokName,
      antrian: noAntrianFormatted,
      waStatus: statusNotif
    };
  } finally {
    lock.releaseLock();
  }
}

    ['AUTO_SEND_TG', cfg.autoTgOnScan ? 'TRUE' : 'FALSE', 'TRUE = kirim notifikasi ke Telegram saat scan']
  ];

  // Update nilai per kunci secara aman atau tambahkan baris baru jika kunci belum ada
  const lastRow = sheet.getLastRow();
  const existingKeys = lastRow > 1
    ? sheet.getRange(2, 1, lastRow - 1, 1).getDisplayValues().map(r => String(r[0]).trim().toUpperCase())
    : [];

function sendWhatsAppGateway(provider, token, targetPhone, message) {
  try {
    const prov = String(provider || 'fonnte').trim().toLowerCase();
    const url = prov.startsWith('http')
      ? provider.trim()
      : (prov.includes('wablas') ? 'https://pati.wablas.com/api/send-message' : 'https://api.fonnte.com/send');
    const res = UrlFetchApp.fetch(url, {
      method: 'post',
      headers: { Authorization: token },
      payload: { target: String(targetPhone), phone: String(targetPhone), message: message },
      muteHttpExceptions: true
    });
    return 'WA (' + res.getResponseCode() + ')';
  } catch (e) {
    return 'Gagal WA: ' + e.message;
  }
}

function sendTelegramBotGateway(botToken, chatId, message, targetPhone, studentName) {
  try {
    const url = 'https://api.telegram.org/bot' + String(botToken).trim() + '/sendMessage';
    let cleanPhone = String(targetPhone || '').replace(/[^0-9]/g, '');
    if (cleanPhone.startsWith('0')) cleanPhone = '62' + cleanPhone.slice(1);

    const payloadObj = {
      chat_id: String(chatId).trim(),
      text: message
    };

    if (cleanPhone) {
      const waDirectUrl = 'https://wa.me/' + cleanPhone + '?text=' + encodeURIComponent(message);
      payloadObj.reply_markup = {
        inline_keyboard: [[
          { text: '📲 Japri WA Wali Murid (' + (studentName || cleanPhone) + ')', url: waDirectUrl }
        ]]
      };
    }

    const res = UrlFetchApp.fetch(url, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(payloadObj),
      muteHttpExceptions: true
    });
    return 'TG (' + res.getResponseCode() + ')';
  } catch (e) {
    return 'Gagal TG: ' + e.message;
  }
}
