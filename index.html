/**
 * PAPAN ANTRIAN BERDAMPINGAN OTOMATIS PER PENGINTERVIEW
 * Membaca daftar nama penginterview langsung dari sheet Data_Penginterview.
 */
function setupAntrianPenginterviewSheet(ss) {
  let qSheet = ss.getSheetByName(SHEET_NAMES.ANTRIAN);
  if (!qSheet) {
    qSheet = ss.insertSheet(SHEET_NAMES.ANTRIAN);
  }
  qSheet.clear();
  qSheet.setTabColor('#0d9488');
  qSheet.setFrozenRows(3);

  // Ambil daftar penginterview unik dari sheet Data_Penginterview
  let interviewerNames = [];
  const pSheet = ss.getSheetByName(SHEET_NAMES.PENGINTERVIEW);
  if (pSheet && pSheet.getLastRow() > 1) {
    const pHeaders = pSheet.getRange(1, 1, 1, pSheet.getLastColumn()).getDisplayValues()[0].map(h => String(h).trim().toLowerCase());
    let nameCol = pHeaders.findIndex(h => /penginterview|pewawancara|interviewer|ustadz|guru|penguji/.test(h));
    if (nameCol === -1) nameCol = pHeaders.length >= 2 ? 1 : 0;
    const pRows = pSheet.getRange(2, nameCol + 1, pSheet.getLastRow() - 1, 1).getDisplayValues();
    pRows.forEach(r => {
      const nm = String(r[0] || '').trim();
      if (nm && interviewerNames.indexOf(nm) === -1) {
        interviewerNames.push(nm);
      }
    });
  }
  if (interviewerNames.length === 0) {
    interviewerNames = ['Ustadz Ahmad Fauzan, S.Pd.', 'Ustadzah Siti Aisyah, S.Pd.I.', 'Ustadz M. Ridwan, M.Pd.'];
  }

  const palette = [
    { headerBg: '#0f766e', subBg: '#f0fdfa' },
    { headerBg: '#4f46e5', subBg: '#eef2ff' },
    { headerBg: '#d97706', subBg: '#fffbeb' },
    { headerBg: '#e11d48', subBg: '#fff1f2' },
    { headerBg: '#0284c7', subBg: '#f0f9ff' },
    { headerBg: '#7c3aed', subBg: '#f5f3ff' }
  ];

  qSheet.setRowHeight(1, 38);
  qSheet.setRowHeight(2, 26);
  qSheet.setRowHeight(3, 30);

  interviewerNames.forEach((name, idx) => {
    const c = (idx * 5) + 1;
    const colors = palette[idx % palette.length];
    const safeName = name.replace(/'/g, "''");
    const safeDoubleQuote = name.replace(/"/g, '""');

    qSheet.getRange(1, c, 1, 4).merge()
      .setValue('🎙️ ' + name.toUpperCase())
      .setBackground(colors.headerBg)
      .setFontColor('#ffffff')
      .setFontWeight('bold')
      .setFontSize(10)
      .setHorizontalAlignment('center')
      .setVerticalAlignment('middle');

    qSheet.getRange(2, c, 1, 2).merge()
      .setValue('Total Antrian:')
      .setBackground(colors.subBg)
      .setFontWeight('bold')
      .setFontSize(9)
      .setHorizontalAlignment('right');
    qSheet.getRange(2, c + 2, 1, 2).merge()
      .setFormula('=COUNTIF(Log_Kehadiran!$M$2:$M, "' + safeDoubleQuote + '") & " Peserta"')
      .setBackground(colors.subBg)
      .setFontWeight('bold')
      .setFontSize(10)
      .setHorizontalAlignment('center');

    qSheet.getRange(3, c, 1, 4)
      .setValues([['No Antrian', 'Nama Siswa', 'Kelompok', 'Waktu Hadir']])
      .setBackground('#1e293b')
      .setFontColor('#ffffff')
      .setFontWeight('bold')
      .setFontSize(9)
      .setHorizontalAlignment('center')
      .setVerticalAlignment('middle');

    // Kolom M = Penginterview, N = No Antrian, G = Nama Siswa, L = Kelompok, C = Waktu Kedatangan
    const formula = '=IFERROR(QUERY(Log_Kehadiran!$A$2:$Q, "SELECT N, G, L, C WHERE M = \'' + safeName + '\' ORDER BY N ASC", 0), {"-", "Belum ada peserta hadir", "-", "-"})';
    qSheet.getRange(4, c).setFormula(formula);

    qSheet.setColumnWidth(c, 85);
    qSheet.setColumnWidth(c + 1, 195);
    qSheet.setColumnWidth(c + 2, 115);
    qSheet.setColumnWidth(c + 3, 100);
    qSheet.setColumnWidth(c + 4, 24);
  });
}

function setupAntrianKelompokSheet(ss) {

/**
 * 2. HANDLER HTTP GET & POST (REST API UNTUK FRONTEND)
 */
function doGet(e) {
  const action = (e && e.parameter && e.parameter.action) || 'ping';
  if (action === 'initData') {
    return jsonResponse(getAllInitialData());
  }
  if (action === 'upgradeSheets') {
    return jsonResponse(syncAndUpgradeSheets());
  }
  if (action === 'scanAttendance' && e.parameter.payload) {
    const payload = JSON.parse(e.parameter.payload);
    return jsonResponse(handleScanAttendance(payload));
  }
  return jsonResponse({ status: 'ok', message: 'E-Presensi QR GAS Backend Active' });
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;

    if (action === 'upgradeSheets') {
      return jsonResponse(syncAndUpgradeSheets());
    }
    if (action === 'scanAttendance') {
      return jsonResponse(handleScanAttendance(payload));
    }
    if (action === 'addSiswa') {
      return jsonResponse(addSiswaRow(payload.siswa));
    }
    if (action === 'addKegiatan') {
      return jsonResponse(addKegiatanRow(payload.kegiatan));
    }
    if (action === 'updateKegiatan') {
      return jsonResponse(updateKegiatanRow(payload.kegiatan));
    }
    if (action === 'setActiveKegiatan') {
      return jsonResponse(setActiveKegiatanRow(payload.idKegiatan));
    }
    if (action === 'saveConfig') {
      return jsonResponse(saveConfigSheet(payload.config));
    }
    return jsonResponse({ status: 'error', message: 'Action tidak dikenali' });
  } catch (err) {
    return jsonResponse({ status: 'error', message: err.toString() });
  }
}

/**
 * 3. PENCATATAN PRESENSI, PENOMORAN ANTRIAN PER PENGINTERVIEW, WA GATEWAY & TELEGRAM BOT
 */
function handleScanAttendance(payload) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let logSheet = ss.getSheetByName(SHEET_NAMES.LOG);
    if (!logSheet) {
      syncAndUpgradeSheets();
      logSheet = ss.getSheetByName(SHEET_NAMES.LOG);
    } else {
      // Pastikan kolom Penginterview tersedia tanpa menghapus data lama
      const curHeaders = logSheet.getRange(1, 1, 1, Math.max(logSheet.getLastColumn(), 1)).getDisplayValues()[0].map(h => String(h).trim().toLowerCase());
      if (curHeaders.indexOf('penginterview') === -1) {
        syncAndUpgradeSheets();
      }
    }

    const kelompokName = String(payload.kelompok || 'Kelompok 1').trim();
    const rawPenginterview = String(payload.penginterview || '').trim();
    const penginterviewName = (rawPenginterview && rawPenginterview !== '-') ? rawPenginterview : kelompokName;
    const idKegiatan = String(payload.idKegiatan || 'KEG-001').trim();

    let queueCounter = 1;
    const lastRow = logSheet.getLastRow();
    if (lastRow > 1) {
      // Kolom D (4) = ID Kegiatan, Kolom L (12) = Kelompok, Kolom M (13) = Penginterview
      const existingLogs = logSheet.getRange(2, 4, lastRow - 1, 10).getDisplayValues();
      for (let i = 0; i < existingLogs.length; i++) {
        const rowKegId = String(existingLogs[i][0]).trim();
        const rowKelompok = String(existingLogs[i][8]).trim();
        const rowPenginterviewRaw = String(existingLogs[i][9] || '').trim();
        const rowKey = (rowPenginterviewRaw && rowPenginterviewRaw !== '-') ? rowPenginterviewRaw : rowKelompok;
        if (rowKegId === idKegiatan && rowKey.toLowerCase() === penginterviewName.toLowerCase()) {
          queueCounter++;
        }
      }
    }
    const noAntrianFormatted = String(queueCounter).padStart(2, '0');

    const now = new Date();
    const waktuWIB = Utilities.formatDate(now, 'Asia/Jakarta', 'HH:mm') + ' WIB';
    const idLog = 'LOG-' + now.getTime();

    const statusChannels = [];
    if (payload.sendWa && payload.waToken && payload.noWa) {
      statusChannels.push(sendWhatsAppGateway(payload.waProvider, payload.waToken, payload.noWa, payload.waMessage));
    }
    if (payload.sendTg && payload.tgToken && payload.tgChatId) {
      const tgText = payload.tgMessage || payload.waMessage;
      statusChannels.push(sendTelegramBotGateway(payload.tgToken, payload.tgChatId, tgText, payload.noWa, payload.namaSiswa, tgText));
    }
    const statusNotif = statusChannels.length > 0 ? statusChannels.join(' | ') : 'Tersimpan di Sheet';

    logSheet.appendRow([
      idLog,
      now.toISOString(),
      payload.waktu || waktuWIB,
      idKegiatan,
      payload.namaKegiatan,
      "'" + payload.nisn,
      payload.namaSiswa,
      payload.panggilan || '',
      payload.jk || 'Laki-laki',
      payload.kelas,
      payload.statusPeserta || 'CALON SISWA',
      kelompokName,
      rawPenginterview || '-',
      "'" + noAntrianFormatted,
      payload.statusKehadiran,
      payload.noWa,
      statusNotif
    ]);

    return {
      status: 'success',
      idLog: idLog,
      waktu: payload.waktu || waktuWIB,
      kelompok: kelompokName,
      penginterview: rawPenginterview || '-',
      antrian: noAntrianFormatted,
      waStatus: statusNotif
    };
  } finally {
    lock.releaseLock();
  }
}

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAMES.KEGIATAN);
  if (!sheet) return { status: 'error', message: 'Sheet kegiatan tidak ditemukan' };
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    const ids = sheet.getRange(2, 1, lastRow - 1, 1).getDisplayValues();
    for (let i = 0; i < ids.length; i++) {
      if (String(ids[i][0]).trim() === String(k.id).trim()) {

        return { status: 'success' };
      }
    }
  }
  return addKegiatanRow(k);
}

function getAllInitialData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const getRows = (name) => {
    const s = ss.getSheetByName(name);
    if (!s || s.getLastRow() <= 1) return [];
    return s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getDisplayValues();
  };
  const getHeaders = (name) => {
    const s = ss.getSheetByName(name);
    if (!s || s.getLastRow() < 1 || s.getLastColumn() < 1) return [];
    return s.getRange(1, 1, 1, s.getLastColumn()).getDisplayValues()[0];
  };
  return {
    status: 'success',
    siswa: getRows(SHEET_NAMES.SISWA),
    kegiatan: getRows(SHEET_NAMES.KEGIATAN),

    penginterviewHeaders: getHeaders(SHEET_NAMES.PENGINTERVIEW),
    penginterview: getRows(SHEET_NAMES.PENGINTERVIEW)
  };
}

function sendWhatsAppGateway(provider, token, targetPhone, message) {
  try {
    const prov = String(provider || 'fonnte').trim().toLowerCase();
    const url = prov.startsWith('http')
      ? provider.trim()
      : (prov.includes('wablas') ? 'https://pati.wablas.com/api/send-message' : 'https://api.fonnte.com/send');
    const res = UrlFetchApp.fetch(url, {
      method: 'post',
      headers: { Authorization: token },
      payload: { target: String(targetPhone), phone: String(targetPhone), message: message },
      muteHttpExceptions: true
    });
    return 'WA (' + res.getResponseCode() + ')';
  } catch (e) {
    return 'Gagal WA: ' + e.message;
  }
}

function normalizePhone62(rawPhone) {
  let digits = String(rawPhone || '').replace(/[^0-9]/g, '');
  if (!digits) return '';
  if (digits.startsWith('0')) digits = '62' + digits.substring(1);
  else if (!digits.startsWith('62')) digits = '62' + digits;
  return digits;
}

function sendTelegramBotGateway(botToken, chatId, message, waPhone, studentName, waCustomText) {
  try {
    const cleanToken = String(botToken || '').trim();
    const url = 'https://api.telegram.org/bot' + cleanToken + '/sendMessage';
    const cleanPhone = normalizePhone62(waPhone);
    const payloadObj = {
      chat_id: String(chatId || '').trim(),
      text: String(message || '')
    };
    if (cleanPhone) {
      const textForWa = String(waCustomText || message || '');
      const waLink = 'https://api.whatsapp.com/send?phone=' + cleanPhone + '&text=' + encodeURIComponent(textForWa);
      payloadObj.reply_markup = {
        inline_keyboard: [[
          { text: '📲 Japri WA Wali Murid (' + (studentName || 'Siswa') + ')', url: waLink }
        ]]
      };
    }
    const res = UrlFetchApp.fetch(url, {
      method: 'post',
      contentType: 'application/json',
      payload: JSON.stringify(payloadObj),
      muteHttpExceptions: true
    });
    return 'TG (' + res.getResponseCode() + ')';
  } catch (e) {
    return 'Gagal TG: ' + e.message;
  }
}
