                            2. Deploy sebagai Web App
                            Klik Deploy → New deployment, pilih jenis Web app, atur Execute as: Me dan Who has access: Anyone.
                        

                        

                            3. Tempel URL di Atas
                            Salin Web App URL hasil deploy lalu tempelkan pada kotak URL Web App GAS tepat di bagian atas halaman ini.

  groups.forEach(g => {
    const c = g.startCol;
    // Banner Nama Kelompok
    const titleRange = qSheet.getRange(1, c, 1, 4);
    titleRange.merge()

/**
 * 2. HANDLER HTTP GET & POST (REST API UNTUK FRONTEND)
 */
function doGet(e) {
  const action = (e && e.parameter && e.parameter.action) || 'ping';
  if (action === 'initData') {
    return jsonResponse(getAllInitialData());
  }
  if (action === 'scanAttendance' && e.parameter.payload) {
    const payload = JSON.parse(e.parameter.payload);
    return jsonResponse(handleScanAttendance(payload));
  }
  return jsonResponse({ status: 'ok', message: 'E-Presensi QR GAS Backend Active' });
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const action = payload.action;

    if (action === 'scanAttendance') {
      return jsonResponse(handleScanAttendance(payload));
    }
    if (action === 'addSiswa') {
      return jsonResponse(addSiswaRow(payload.siswa));
    }
    if (action === 'addKegiatan') {
      return jsonResponse(addKegiatanRow(payload.kegiatan));
    }
    if (action === 'updateKegiatan') {
      return jsonResponse(updateKegiatanRow(payload.kegiatan));
    }
    if (action === 'setActiveKegiatan') {
      return jsonResponse(setActiveKegiatanRow(payload.idKegiatan));
    }
    if (action === 'saveConfig') {
      return jsonResponse(saveConfigSheet(payload.config));
    }
    return jsonResponse({ status: 'error', message: 'Action tidak dikenali' });
  } catch (err) {
    return jsonResponse({ status: 'error', message: err.toString() });
  }
}

/**
 * 3. PENCATATAN PRESENSI & PENOMORAN ANTRIAN 2 DIGIT OTOMATIS PER KELOMPOK
 */
function handleScanAttendance(payload) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);

  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let logSheet = ss.getSheetByName(SHEET_NAMES.LOG);
    if (!logSheet) {
      initializeSheets();
      logSheet = ss.getSheetByName(SHEET_NAMES.LOG);
    }

    const kelompokName = String(payload.kelompok || 'Kelompok 1').trim();
    const idKegiatan = String(payload.idKegiatan || 'KEG-001').trim();

    // Hitung urutan nomor antrian pada Kegiatan & Kelompok yang sama
    let queueCounter = 1;
    const lastRow = logSheet.getLastRow();
    if (lastRow > 1) {
      const existingLogs = logSheet.getRange(2, 4, lastRow - 1, 9).getDisplayValues();
      for (let i = 0; i < existingLogs.length; i++) {
        const rowKegId = String(existingLogs[i][0]).trim();
        const rowKelompok = String(existingLogs[i][8]).trim();
        if (rowKegId === idKegiatan && rowKelompok.toLowerCase() === kelompokName.toLowerCase()) {
          queueCounter++;
        }
      }
    }
    const noAntrianFormatted = String(queueCounter).padStart(2, '0');

    const now = new Date();
    const waktuWIB = Utilities.formatDate(now, 'Asia/Jakarta', 'HH:mm') + ' WIB';
    const idLog = 'LOG-' + now.getTime();

    let statusNotif = 'Terkirim via Antrean';
    if (payload.sendWa && payload.waToken && payload.noWa) {
      statusNotif = sendWhatsAppGateway(payload.waProvider, payload.waToken, payload.noWa, payload.waMessage);
    }

    logSheet.appendRow([
      idLog,
      now.toISOString(),
      payload.waktu || waktuWIB,
      idKegiatan,
      payload.namaKegiatan,
      "'" + payload.nisn,
      payload.namaSiswa,
      payload.panggilan || '',
      payload.jk || 'Laki-laki',
      payload.kelas,
      payload.statusPeserta || 'CALON SISWA',
      kelompokName,
      "'" + noAntrianFormatted,
      payload.statusKehadiran,
      payload.noWa,
      statusNotif
    ]);

    return {
      status: 'success',
      idLog: idLog,
      waktu: payload.waktu || waktuWIB,
      kelompok: kelompokName,
      antrian: noAntrianFormatted,
      waStatus: statusNotif
    };
  } finally {
    lock.releaseLock();
  }
}

/**
 * 4. CRUD TAMBAH SISWA, KEGIATAN & KONFIGURASI KE GOOGLE SHEET
 */
function addSiswaRow(s) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAMES.SISWA);
  if (!sheet) { initializeSheets(); sheet = ss.getSheetByName(SHEET_NAMES.SISWA); }
  const qrUrl = 'https://quickchart.io/qr?text=' + encodeURIComponent(s.nisn) + '&size=300&margin=2';
  sheet.appendRow([
    "'" + s.nisn,
    s.nama,
    s.panggilan || '',
    s.jk || 'Laki-laki',
    s.kelas,
    s.statusPeserta || 'CALON SISWA',
    s.kelompok || 'Kelompok 1',
    s.wa,
    qrUrl,
    'Aktif'
  ]);
  return { status: 'success' };
}

function addKegiatanRow(k) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAMES.KEGIATAN);
  if (!sheet) { initializeSheets(); sheet = ss.getSheetByName(SHEET_NAMES.KEGIATAN); }
  sheet.appendRow([k.id, k.nama, k.tanggal, k.mulai, k.batasTerlambat, k.lokasi, k.status || 'Aktif']);
  return { status: 'success' };
}

function updateKegiatanRow(k) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAMES.KEGIATAN);
  if (!sheet) return { status: 'error', message: 'Sheet kegiatan tidak ditemukan' };
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    const ids = sheet.getRange(2, 1, lastRow - 1, 1).getDisplayValues();
    for (let i = 0; i < ids.length; i++) {
      if (String(ids[i][0]).trim() === String(k.id).trim()) {
        sheet.getRange(i + 2, 1, 1, 7).setValues([[k.id, k.nama, k.tanggal, k.mulai, k.batasTerlambat, k.lokasi, k.status || 'Aktif']]);
        return { status: 'success' };
      }
    }
  }
  return addKegiatanRow(k);
}

function setActiveKegiatanRow(activeId) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAMES.KEGIATAN);
  if (!sheet) return { status: 'error' };
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) {
    const data = sheet.getRange(2, 1, lastRow - 1, 7).getDisplayValues();
    for (let i = 0; i < data.length; i++) {
      const statusVal = String(data[i][0]).trim() === String(activeId).trim() ? 'Aktif' : 'Terjadwal';
      sheet.getRange(i + 2, 7).setValue(statusVal);
    }
  }
  return { status: 'success' };
}

    ['TPL_NOTIF_HADIR', cfg.tplNotifHadir || '', 'Template WA saat scan berhasil']
  ];
  sheet.getRange(2, 1, rows.length, 3).setValues(rows);
  return { status: 'success' };
}

function getAllInitialData() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const getRows = (name) => {
    const s = ss.getSheetByName(name);
    if (!s || s.getLastRow() <= 1) return [];
    return s.getRange(2, 1, s.getLastRow() - 1, s.getLastColumn()).getDisplayValues();
  };
  return {
    status: 'success',
    siswa: getRows(SHEET_NAMES.SISWA),
    kegiatan: getRows(SHEET_NAMES.KEGIATAN),
    logs: getRows(SHEET_NAMES.LOG),
    config: getRows(SHEET_NAMES.CONFIG)
  };
}

function sendWhatsAppGateway(provider, token, targetPhone, message) {
  try {
    const prov = String(provider || 'fonnte').trim().toLowerCase();
    const url = prov.startsWith('http')
      ? provider.trim()
      : (prov.includes('wablas') ? 'https://pati.wablas.com/api/send-message' : 'https://api.fonnte.com/send');
    const res = UrlFetchApp.fetch(url, {
      method: 'post',
      headers: { Authorization: token },
      payload: { target: String(targetPhone), phone: String(targetPhone), message: message },
      muteHttpExceptions: true
    });
    return 'Terkirim (' + res.getResponseCode() + ')';
  } catch (e) {
    return 'Gagal WA: ' + e.message;
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
